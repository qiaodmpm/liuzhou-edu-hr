> 版本：v1.0 | 日期：2026-06-24 | 状态：方案确认中，暂不改代码 | 关联产品方案：`产品方案_未使用编制补录.md`

## 1. 总体架构

### 1.1 核心思路

在台账数据结构中新增 `careerPlans[]` / `controlPlans[]` 数组，替代单纯的聚合数值。每条计划记录 = 一次编制批复的完整档案。聚合字段保留但改为从数组自动计算。

### 2.1 `src/ledger-plan-utils.js` — 计划记录工具函数库

IIFE 模式，挂载到 `window.LedgerPlanUtils`，与项目现有模式一致。

```javascript
/**
 * 编制岗位台账 — 计划记录工具函数库
 * IIFE, window.LedgerPlanUtils
 * 纯函数集合，不直接操作 DOM 或 localStorage
 */
(function () {
  'use strict';

  // ============================================================
  // 常量
  // ============================================================

  /** 计划来源 */
  const PLAN_SOURCE = { ONLINE: 'online', OFFLINE: 'offline' };

  /** 计划状态 */
  const PLAN_STATUS = {
    ACTIVE: 'active',         // 有效
    EXHAUSTED: 'exhausted',   // 已用完
    EXPIRED: 'expired',       // 已到期
    ADJUSTED: 'adjusted'      // 已核减调整
  };

  /** 到期预警天数 */
  const EXPIRE_WARN_DAYS = 60;

  // ============================================================
  // 计划记录 CRUD
  // ============================================================

  /**
   * 生成计划编号
   * @param {'C'|'K'} track - C=事业编, K=控制数
   * @param {string} [dateStr] - 日期 YYYYMMDD，默认今天
   * @returns {string} PLAN-C-20260624001
   */
  function generatePlanId(track, dateStr) {
    const d = dateStr || new Date().toISOString().slice(0, 10).replace(/-/g, '');
    const seq = String(Math.floor(Math.random() * 900) + 100); // 3位随机序号，实际应从已有最大+1
    return `PLAN-${track}-${d}${seq}`;
  }

  /**
   * 创建计划记录（初始模板）
   * @param {object} opts
   * @returns {object} 计划记录
   */
  function createPlanRecord(opts) {
    const now = new Date().toISOString().slice(0, 10);
    return {
      id: opts.id || generatePlanId(opts.track || 'C'),
      source: opts.source || 'offline',          // 'online' | 'offline'
      // 线上来源字段
      sourceBatchId: opts.sourceBatchId || '',
      sourceSubmissionId: opts.sourceSubmissionId || '',
      // 线下来源字段
      offlineDocNo: opts.offlineDocNo || '',
      offlineApprovedBy: opts.offlineApprovedBy || '',
      // 批复内容
      approvedCount: opts.approvedCount || 0,
      adjustedApprovedCount: opts.adjustedApprovedCount ?? opts.approvedCount ?? 0,
      usedCount: opts.usedCount || 0,
      deferredCount: opts.deferredCount || 0,
      // 有效期
      validStart: opts.validStart || '',
      validEnd: opts.validEnd || '',
      // 状态（自动判定）
      status: PLAN_STATUS.ACTIVE,
      // 使用类型明细
      useTypes: opts.useTypes || {
        publicRecruit: { approved: 0, used: 0 },
        selfRecruit:   { approved: 0, used: 0 },
        talentMid:     { approved: 0, used: 0 },
        talentHigh:    { approved: 0, used: 0 },
        transfer:      { approved: 0, used: 0 },
        other:         { approved: 0, used: 0 }
      },
      // 关联聘用（用于核减/暂缓追溯）
      linkedEmployIds: opts.linkedEmployIds || [],
      // 元数据
      createdAt: opts.createdAt || now,
      createdBy: opts.createdBy || '',
      updatedAt: now,
      notes: opts.notes || ''
    };
  }

  // ============================================================
  // 计算字段
  // ============================================================

  /**
   * 计算计划剩余可用名额
   * @param {object} plan
   * @returns {number}
   */
  function remainingCount(plan) {
    return Math.max(0, plan.adjustedApprovedCount - plan.usedCount - plan.deferredCount);
  }

  /**
   * 判定计划状态（基于日期和数值）
   * @param {object} plan
   * @param {string} [today] - 参考日期，默认今天
   * @returns {string} PLAN_STATUS 之一
   */
  function computePlanStatus(plan, today) {
    const ref = today || new Date().toISOString().slice(0, 10);
    if (plan.validEnd && plan.validEnd < ref && remainingCount(plan) > 0) {
      return PLAN_STATUS.EXPIRED;
    }
    if (remainingCount(plan) <= 0) {
      return PLAN_STATUS.EXHAUSTED;
    }
    if (plan.adjustedApprovedCount < plan.approvedCount) {
      return PLAN_STATUS.ADJUSTED;
    }
    return PLAN_STATUS.ACTIVE;
  }

  /**
   * 刷新计划状态（原地修改 + 返回）
   * @param {object} plan
   * @param {string} [today]
   * @returns {object} plan
   */
  function refreshStatus(plan, today) {
    plan.status = computePlanStatus(plan, today);
    return plan;
  }

  /**
   * 是否即将到期
   * @param {object} plan
   * @param {string} [today]
   * @returns {boolean}
   */
  function isExpiringSoon(plan, today) {
    if (!plan.validEnd) return false;
    const ref = today || new Date().toISOString().slice(0, 10);
    const daysLeft = Math.ceil((new Date(plan.validEnd) - new Date(ref)) / 86400000);
    return daysLeft > 0 && daysLeft <= EXPIRE_WARN_DAYS && remainingCount(plan) > 0;
  }

  // ============================================================
  // 聚合计算（从 plans 数组计算台账聚合字段）
  // ============================================================

  /**
   * 从计划记录数组计算聚合值
   * @param {object[]} plans
   * @param {string} [today]
   * @returns {{ approvedPlan: number, usedPlan: number, unusedPlan: number, unusedActive: number, unusedExpired: number }}
   */
  function computeAggregates(plans, today) {
    if (!plans || plans.length === 0) {
      return { approvedPlan: 0, usedPlan: 0, unusedPlan: 0, unusedActive: 0, unusedExpired: 0 };
    }
    const ref = today || new Date().toISOString().slice(0, 10);

    let approvedPlan = 0;   // 有效期内 + active/adjusted 状态的 adjustedApprovedCount
    let usedPlan = 0;
    let unusedActive = 0;
    let unusedExpired = 0;

    for (const p of plans) {
      const st = computePlanStatus(p, ref);
      usedPlan += p.usedCount;

      if (st === PLAN_STATUS.EXPIRED) {
        unusedExpired += remainingCount(p);
      } else {
        // active / exhausted / adjusted 都计入已批复
        approvedPlan += p.adjustedApprovedCount;
        if (st !== PLAN_STATUS.EXHAUSTED) {
          unusedActive += remainingCount(p);
        }
      }
    }

    return {
      approvedPlan,
      usedPlan,
      unusedPlan: unusedActive + unusedExpired,
      unusedActive,
      unusedExpired
    };
  }

  /**
   * 计算可申请编制数
   * @param {number} quota - 核定数
   * @param {number} actual - 实有在编人数
   * @param {object[]} plans - 计划记录数组
   * @param {string} [today]
   * @returns {number}
   */
  function computeAvailable(quota, actual, plans, today) {
    const agg = computeAggregates(plans, today);
    return Math.max(0, quota - actual - agg.approvedPlan);
  }

  // ============================================================
  // 联动操作
  // ============================================================

  /**
   * 扣减计划记录已使用数（入编通过时调用）
   * @param {object} plan - 原地修改
   * @param {number} count
   * @param {string} [entryId] - 入编记录 ID，用于追溯
   * @returns {object} plan
   */
  function consumePlan(plan, count, entryId) {
    plan.usedCount += count;
    plan.updatedAt = new Date().toISOString().slice(0, 10);
    if (entryId) {
      plan.notes = (plan.notes ? plan.notes + '; ' : '') + `入编(${entryId}) +${count}人`;
    }
    return refreshStatus(plan);
  }

  /**
   * 聘用核减联动：扣减 adjustedApprovedCount
   * @param {object} plan - 原地修改
   * @param {number} cancelCount - 核减人数
   * @param {object} context - 上下文信息
   * @param {string} context.employId - 聘用记录 ID
   * @param {string} context.positionName - 岗位名称
   * @param {string} context.cancelDocNo - 核减依据文号
   * @returns {{ success: boolean, error?: string }}
   */
  function applyCancelReduction(plan, cancelCount, context) {
    if (cancelCount <= 0) return { success: true };

    const remaining = remainingCount(plan);
    if (cancelCount > plan.adjustedApprovedCount) {
      return { success: false, error: `核减数(${cancelCount})超过已批复计划数(${plan.adjustedApprovedCount})` };
    }

    plan.adjustedApprovedCount -= cancelCount;
    plan.updatedAt = new Date().toISOString().slice(0, 10);
    plan.notes = (plan.notes ? plan.notes + '; ' : '')
      + `聘用(${context.employId})核减${cancelCount}人`
      + (context.positionName ? `(岗位:${context.positionName})` : '')
      + (context.cancelDocNo ? `,依据:${context.cancelDocNo}` : '');

    if (!plan.linkedEmployIds.includes(context.employId)) {
      plan.linkedEmployIds.push(context.employId);
    }

    return { success: true, plan: refreshStatus(plan) };
  }

  /**
   * 聘用暂缓联动：累加 deferredCount
   * @param {object} plan - 原地修改
   * @param {number} deferredCount
   * @param {object} context
   * @param {string} context.employId
   * @param {Array<{name:string, reason:string}>} context.persons - 暂缓人员
   * @returns {{ success: boolean, error?: string }}
   */
  function applyDeferred(plan, deferredCount, context) {
    if (deferredCount <= 0) return { success: true };

    const remaining = remainingCount(plan);
    if (deferredCount > remaining) {
      return { success: false, error: `暂缓人数(${deferredCount})超过剩余可用名额(${remaining})` };
    }

    plan.deferredCount += deferredCount;
    plan.updatedAt = new Date().toISOString().slice(0, 10);

    const names = (context.persons || []).map(p => p.name || '(岗位级暂缓)').join(',');
    plan.notes = (plan.notes ? plan.notes + '; ' : '')
      + `聘用(${context.employId})暂缓${deferredCount}人:${names}`;

    if (!plan.linkedEmployIds.includes(context.employId)) {
      plan.linkedEmployIds.push(context.employId);
    }

    return { success: true, plan: refreshStatus(plan) };
  }

  /**
   * 反向恢复核减/暂缓（聘用驳回重提时）
   * @param {object} plan - 原地修改
   * @param {number} cancelCount - 要恢复的核减数
   * @param {number} deferredCount - 要恢复的暂缓数
   * @returns {object} plan
   */
  function revertPlanAdjustment(plan, cancelCount, deferredCount) {
    plan.adjustedApprovedCount += cancelCount;
    plan.deferredCount = Math.max(0, plan.deferredCount - deferredCount);
    plan.updatedAt = new Date().toISOString().slice(0, 10);
    plan.notes = (plan.notes ? plan.notes + '; ' : '') + '聘用驳回重提，恢复核减/暂缓';
    return refreshStatus(plan);
  }

  // ============================================================
  // 链路追踪（聘用 → 招聘 → 编制 → 台账计划记录）
  // ============================================================

  /**
   * 从聘用记录追溯到台账计划记录
   * @param {object} employRecord - 聘用记录
   * @param {object[]} allRecruitRecords - 全部招聘记录
   * @param {object[]} allStaffingSubmissions - 全部编制申请
   * @param {object[]} careerPlans - 台账事业编计划记录
   * @param {object[]} controlPlans - 台账控制数计划记录
   * @returns {{ plan: object|null, planType: 'career'|'control'|null, error?: string }}
   */
  function traceToPlan(employRecord, allRecruitRecords, allStaffingSubmissions, careerPlans, controlPlans) {
    // Step 1: 聘用 → 招聘批次
    const recruit = allRecruitRecords.find(r => r.id === employRecord.recruitBatchId);
    if (!recruit) {
      return { plan: null, planType: null, error: '关联的招聘批次不存在' };
    }

    // Step 2: 招聘 → 编制申请
    const staffingBatchId = recruit.staffingBatchId;
    if (!staffingBatchId) {
      return { plan: null, planType: null, error: '招聘批次未关联编制使用申请批次' };
    }

    // 找到该编制批次下所有已完成的申请
    const staffingSubs = allStaffingSubmissions.filter(s =>
      s.batchId === staffingBatchId && s.status === 'approved'
    );
    if (staffingSubs.length === 0) {
      return { plan: null, planType: null, error: '关联的编制申请不存在或未完成' };
    }

    // Step 3: 编制申请 → 台账计划记录
    // 一个编制批次可能有多条申请（事业编+控制数），取第一条匹配的
    for (const sub of staffingSubs) {
      const plans = sub.staffingType === 'career' ? careerPlans : controlPlans;
      const plan = plans.find(p => p.sourceSubmissionId === sub.id);
      if (plan) {
        return { plan, planType: sub.staffingType };
      }
    }

    return { plan: null, planType: null, error: '台账中未找到对应的计划记录' };
  }

  // ============================================================
  // 校验
  // ============================================================

  /**
   * 校验补录数据
   * @param {object} data
   * @param {object[]} existingPlans - 已有计划记录（用于查重）
   * @returns {{ valid: boolean, errors: string[] }}
   */
  function validateBackfill(data, existingPlans) {
    const errors = [];
    if (!data.offlineDocNo || !data.offlineDocNo.trim()) errors.push('请填写批复文号');
    if (existingPlans && existingPlans.some(p => p.offlineDocNo === data.offlineDocNo)) {
      errors.push('该文号已存在');
    }
    if (!data.offlineApprovedBy || !data.offlineApprovedBy.trim()) errors.push('请填写批复机关');
    if (!data.approvedCount || data.approvedCount < 1) errors.push('请填写批复人数（≥1）');
    if (data.usedCount < 0 || data.usedCount > data.approvedCount) errors.push('已使用人数不能超过批复人数');
    if (!data.validStart) errors.push('请选择有效期起始');
    if (!data.validEnd) errors.push('请选择有效期截止');
    if (data.validStart && data.validEnd && data.validEnd < data.validStart) {
      errors.push('截止日期不能早于起始日期');
    }
    // 使用类型合计校验
    if (data.useTypes) {
      const typeSum = Object.values(data.useTypes).reduce((s, t) => s + (t.approved || 0), 0);
      if (typeSum > 0 && typeSum !== data.approvedCount) {
        errors.push(`使用类型合计(${typeSum})不等于批复人数(${data.approvedCount})`);
      }
    }
    return { valid: errors.length === 0, errors };
  }

  /**
   * 校验核减联动
   * @param {object} employRecord
   * @param {object[]} allRecruitRecords
   * @param {object[]} allStaffingSubmissions
   * @param {object[]} careerPlans
   * @param {object[]} controlPlans
   * @returns {{ valid: boolean, errors: string[] }}
   */
  function validateCancelLinkage(employRecord, allRecruitRecords, allStaffingSubmissions, careerPlans, controlPlans) {
    const errors = [];
    const traceResult = traceToPlan(employRecord, allRecruitRecords, allStaffingSubmissions, careerPlans, controlPlans);
    if (traceResult.error) {
      errors.push(traceResult.error);
      return { valid: false, errors };
    }

    // 汇总各岗位核减数
    const positions = employRecord.formData?.recruitmentForms?.positions || [];
    const totalCancel = positions.reduce((s, p) => s + (p.cancelCount || 0), 0);
    const totalDeferred = positions.reduce((s, p) => s + (p.deferredPersons || []).length, 0);

    if (totalCancel > traceResult.plan.adjustedApprovedCount) {
      errors.push(`核减总数(${totalCancel})超过已批复计划可用数(${traceResult.plan.adjustedApprovedCount})`);
    }
    if (totalDeferred > remainingCount(traceResult.plan)) {
      errors.push(`暂缓人数(${totalDeferred})超过剩余可用名额(${remainingCount(traceResult.plan)})`);
    }

    return { valid: errors.length === 0, errors, traceResult, totalCancel, totalDeferred };
  }

  // ============================================================
  // 导出
  // ============================================================

  window.LedgerPlanUtils = {
    PLAN_SOURCE,
    PLAN_STATUS,
    EXPIRE_WARN_DAYS,
    generatePlanId,
    createPlanRecord,
    remainingCount,
    computePlanStatus,
    refreshStatus,
    isExpiringSoon,
    computeAggregates,
    computeAvailable,
    consumePlan,
    applyCancelReduction,
    applyDeferred,
    revertPlanAdjustment,
    traceToPlan,
    validateBackfill,
    validateCancelLinkage
  };

})();
```

### 2.2 `src/ledger-migrate.js` — 数据迁移脚本

IIFE 模式，挂载到 `window.LedgerMigrate`。

```javascript
/**
 * 编制岗位台账 — 数据迁移脚本
 * IIFE, window.LedgerMigrate
 * 负责将旧格式台账数据迁移至新格式（含 careerPlans / controlPlans）
 */
(function () {
  'use strict';

  const LPU = window.LedgerPlanUtils;
  if (!LPU) {
    console.error('[LedgerMigrate] LedgerPlanUtils not loaded');
    return;
  }

  /**
   * 迁移单个台账从旧格式到新格式
   * @param {object} ledger - 台账对象（原地修改）
   * @param {object[]} [allStaffingSubmissions] - 全部编制申请（用于自动生成线上计划记录）
   * @returns {{ migrated: boolean, pendingBackfill?: {career:number, control:number} }}
   */
  function migrateLedger(ledger, allStaffingSubmissions) {
    // 已有 plans 数组 → 无需迁移
    if (ledger.careerPlans && Array.isArray(ledger.careerPlans)) {
      // 仅刷新状态
      ledger.careerPlans.forEach(p => LPU.refreshStatus(p));
      if (ledger.controlPlans) ledger.controlPlans.forEach(p => LPU.refreshStatus(p));
      return { migrated: false };
    }

    const submissions = allStaffingSubmissions || [];
    const schoolSubs = submissions.filter(s => s.school === ledger.schoolName && s.status === 'approved');

    // 迁移事业编
    ledger.careerPlans = schoolSubs
      .filter(s => s.staffingType === 'career')
      .map(s => ({
        id: 'PLAN-C-' + s.id,
        source: 'online',
        sourceBatchId: s.batchId || '',
        sourceSubmissionId: s.id,
        offlineDocNo: '',
        offlineApprovedBy: '',
        approvedCount: s.finalApprovedCount || s.totalStaff || 0,
        adjustedApprovedCount: s.finalApprovedCount || s.totalStaff || 0,
        usedCount: 0,
        deferredCount: 0,
        validStart: s.validStart || '',
        validEnd: s.validEnd || '',
        status: LPU.computePlanStatus({
          adjustedApprovedCount: s.finalApprovedCount || s.totalStaff || 0,
          usedCount: 0,
          deferredCount: 0,
          approvedCount: s.finalApprovedCount || s.totalStaff || 0,
          validEnd: s.validEnd || ''
        }),
        useTypes: migrateUseTypes(s.useForms, s.finalApprovedDetails),
        linkedEmployIds: [],
        createdAt: s.approveDate || s.submitDate || '',
        createdBy: '系统迁移',
        updatedAt: ledger.updatedAt || new Date().toISOString().slice(0, 10),
        notes: '从编制使用申请记录自动迁移'
      }));

    // 迁移控制数
    ledger.controlPlans = schoolSubs
      .filter(s => s.staffingType === 'control')
      .map(s => ({
        id: 'PLAN-K-' + s.id,
        source: 'online',
        sourceBatchId: s.batchId || '',
        sourceSubmissionId: s.id,
        offlineDocNo: '',
        offlineApprovedBy: '',
        approvedCount: s.finalApprovedCount || 0,
        adjustedApprovedCount: s.finalApprovedCount || 0,
        usedCount: 0,
        deferredCount: 0,
        validStart: s.validStart || '',
        validEnd: s.validEnd || '',
        status: LPU.computePlanStatus({
          adjustedApprovedCount: s.finalApprovedCount || 0,
          usedCount: 0,
          deferredCount: 0,
          approvedCount: s.finalApprovedCount || 0,
          validEnd: s.validEnd || ''
        }),
        useTypes: migrateUseTypes(s.useForms, s.finalApprovedDetails),
        linkedEmployIds: [],
        createdAt: s.approveDate || s.submitDate || '',
        createdBy: '系统迁移',
        updatedAt: ledger.updatedAt || new Date().toISOString().slice(0, 10),
        notes: '从编制使用申请记录自动迁移'
      }));

    // 检测差额（需要补录的部分）
    const careerOnlineTotal = ledger.careerPlans.reduce((s, p) => s + p.adjustedApprovedCount, 0);
    const controlOnlineTotal = ledger.controlPlans.reduce((s, p) => s + p.adjustedApprovedCount, 0);

    const pendingBackfill = {};
    if (careerOnlineTotal < (ledger.careerApprovedPlan || 0)) {
      pendingBackfill.career = ledger.careerApprovedPlan - careerOnlineTotal;
    }
    if (controlOnlineTotal < (ledger.controlApprovedPlan || 0)) {
      pendingBackfill.control = ledger.controlApprovedPlan - controlOnlineTotal;
    }

    // 标记待补录
    if (pendingBackfill.career || pendingBackfill.control) {
      ledger._pendingBackfill = pendingBackfill;
    }

    // 从 plans 重新计算聚合值
    const careerAgg = LPU.computeAggregates(ledger.careerPlans);
    ledger.careerApprovedPlan = careerAgg.approvedPlan;
    ledger.careerUsedPlan = careerAgg.usedPlan;

    const controlAgg = LPU.computeAggregates(ledger.controlPlans);
    ledger.controlApprovedPlan = controlAgg.approvedPlan;
    ledger.controlUsedPlan = controlAgg.usedPlan;

    return { migrated: true, pendingBackfill: (pendingBackfill.career || pendingBackfill.control) ? pendingBackfill : null };
  }

  /**
   * 迁移 useForms → useTypes
   */
  function migrateUseTypes(useForms, finalApprovedDetails) {
    const types = {
      publicRecruit: { approved: 0, used: 0 },
      selfRecruit:   { approved: 0, used: 0 },
      talentMid:     { approved: 0, used: 0 },
      talentHigh:    { approved: 0, used: 0 },
      transfer:      { approved: 0, used: 0 },
      other:         { approved: 0, used: 0 }
    };
    if (useForms) {
      for (const key of Object.keys(types)) {
        if (useForms[key]) {
          types[key].approved = useForms[key].num || 0;
        }
      }
    }
    if (finalApprovedDetails) {
      for (const d of finalApprovedDetails) {
        const key = d.formType;
        if (types[key]) {
          types[key].approved = d.finalNum || types[key].approved;
        }
      }
    }
    return types;
  }

  /**
   * 检查台账是否需要迁移
   * @param {object} ledger
   * @returns {boolean}
   */
  function needsMigration(ledger) {
    return !ledger.careerPlans || !Array.isArray(ledger.careerPlans);
  }

  /**
   * 递归检查所有台账（用于 org_ledger 存储）
   * @param {object} orgLedger - { [schoolName]: SchoolLedger }
   * @param {object[]} [submissions]
   * @returns {{ migrated: string[], pendingBackfill: object }}
   */
  function migrateAll(orgLedger, submissions) {
    const migrated = [];
    const pendingBackfill = {};
    for (const [schoolName, ledger] of Object.entries(orgLedger)) {
      const result = migrateLedger(ledger, submissions);
      if (result.migrated) migrated.push(schoolName);
      if (result.pendingBackfill) pendingBackfill[schoolName] = result.pendingBackfill;
    }
    return { migrated, pendingBackfill };
  }

  window.LedgerMigrate = {
    migrateLedger,
    needsMigration,
    migrateAll
  };

})();
```

```javascript
// 台账侧监听聘用完成事件
window.addEventListener('employ:completed', function (e) {
  const { employRecord } = e.detail;
  executeCancelDeferredLinkage(employRecord);
});

function executeCancelDeferredLinkage(employRecord) {
  const LPU = window.LedgerPlanUtils;

  // 读取全部数据
  const allRecruitRecords = JSON.parse(localStorage.getItem('recruit_records') || '[]');
  const allStaffingSubmissions = JSON.parse(localStorage.getItem('staffing_submissions') || '[]');
  const orgLedger = JSON.parse(localStorage.getItem('org_ledger') || '{}');
  const school = orgLedger[employRecord.schoolName];
  if (!school) return;

  // 链路追踪
  const traceResult = LPU.traceToPlan(
    employRecord, allRecruitRecords, allStaffingSubmissions,
    school.careerPlans || [], school.controlPlans || []
  );

  if (traceResult.error) {
    console.warn('[EmployLinkage]', traceResult.error);
    return;
  }

  const { plan } = traceResult;

  // 汇总各岗位的核减和暂缓
  const positions = employRecord.formData?.recruitmentForms?.positions || [];
  for (const pos of positions) {
    // 核减
    if (pos.cancelCount > 0) {
      const result = LPU.applyCancelReduction(plan, pos.cancelCount, {
        employId: employRecord.id,
        positionName: pos.positionName,
        cancelDocNo: pos.cancelDocNo
      });
      if (!result.success) {
        console.warn('[EmployLinkage] 核减失败:', result.error);
      }
    }

    // 暂缓
    const deferredCount = (pos.deferredPersons || []).length;
    if (deferredCount > 0) {
      LPU.applyDeferred(plan, deferredCount, {
        employId: employRecord.id,
        persons: pos.deferredPersons
      });
    }
  }

  // 重新计算聚合值
  const track = traceResult.planType === 'career' ? 'career' : 'control';
  const plans = school[track + 'Plans'];
  const agg = LPU.computeAggregates(plans);
  school[track + 'ApprovedPlan'] = agg.approvedPlan;
  school[track + 'UsedPlan'] = agg.usedPlan;

  // 写回
  orgLedger[employRecord.schoolName] = school;
  localStorage.setItem('org_ledger', JSON.stringify(orgLedger));

  console.log('[EmployLinkage] 核减/暂缓联动完成, plan:', plan.id);
}
```
